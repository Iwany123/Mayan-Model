Mayan Architectural Style Classifier

This project implements a Convolutional Neural Network (CNN) to classify images of Mayan architecture into three distinct styles: Calak Mul, Peten, and PUUC.

Project Goal

The primary goal is to accurately identify the architectural style of a given Mayan structure from an image input. This classification model serves as a foundational step towards potentially exploring generative models for creating new architectural styles.

Model Architecture

The classification model is a sequential CNN built using TensorFlow/Keras, featuring convolutional layers for feature extraction, max-pooling for dimensionality reduction, and dense layers for final classification. Dropout is included to mitigate overfitting.

How to Use the Model (Python Script)

Below is the consolidated Python script (`mayan_classifier.py`) that encapsulates the model's definition, loading mechanism, and a prediction function. To use it:

  1. Save the following code as mayan_classifier.py in your local directory.
  2. Ensure you have the necessary libraries installed: pip install tensorflow numpy Pillow scikit-learn
  3. You will need two files generated during training: mayan_style_classifier.h5 (your trained model) and label_encoder.pkl (the label mapping). Place these in the same directory as mayan_classifier.py.
  4. Run the script from your terminal: python mayan_classifier.py. It will demonstrate how to load the model and make a prediction on a dummy image.
  5. Modify the demo_image_path in the if __name__ == "__main__": block to point to your own image for classification.

mayan_classifier.py Code


import os
import numpy as np
from PIL import Image
import pickle
from tensorflow.keras.models import Sequential, load_model
from tensorflow.keras.layers import Conv2D, MaxPooling2D, Flatten, Dense, Dropout
from tensorflow.keras.optimizers import Adam

# --- 1. Model Definition ---
def build_model(input_shape=(224, 224, 3), num_classes=3):
    """Defines and compiles the CNN model for Mayan style classification."""
    model = Sequential([
        Conv2D(32, (3, 3), activation='relu', input_shape=input_shape),
        MaxPooling2D((2, 2)),
        Conv2D(64, (3, 3), activation='relu'),
        MaxPooling2D((2, 2)),
        Conv2D(128, (3, 3), activation='relu'),
        MaxPooling2D((2, 2)),
        Flatten(),
        Dense(128, activation='relu'),
        Dropout(0.5),
        Dense(num_classes, activation='softmax')
    ])
    model.compile(
        optimizer=Adam(learning_rate=0.001),
        loss='categorical_crossentropy',
        metrics=['accuracy']
    )
    return model

# --- 2. Saving the Trained Model and Label Encoder (for reference) ---
# In your training notebook, after training the 'model' and 'label_encoder':

# with open(label_encoder_filename, 'wb') as f:

# --- 3. Loading the Model and Label Encoder ---
def load_classifier(model_path='mayan_style_classifier.h5', label_encoder_path='label_encoder.pkl'):
    """Loads the pre-trained model and label encoder."""
    try:
        loaded_model = load_model(model_path)
        print(f"Model loaded successfully from {model_path}!")

        with open(label_encoder_path, 'rb') as f:
            loaded_label_encoder = pickle.load(f)
        print(f"Label encoder loaded successfully from {label_encoder_path}!")
        return loaded_model, loaded_label_encoder
    except Exception as e:
        print(f"Error loading classifier components: {e}")
        return None, None

# --- 4. Prediction Function ---
def predict_style(image_path, model, label_encoder, img_size=(224, 224)):
    """Predicts the architectural style of a given image."""
    if model is None or label_encoder is None:
        return "Error: Model or label encoder not loaded.", None

    try:
        img = Image.open(image_path).convert('RGB') # Ensure 3 channels
        img = img.resize(img_size) # Resize image to expected input size
        img_array = np.array(img) / 255.0  # Normalize pixel values (0-1)
        img_array = np.expand_dims(img_array, axis=0) # Add batch dimension

        # Make prediction
        predictions = model.predict(img_array)
        predicted_class_index = np.argmax(predictions, axis=1)[0]
        predicted_style = label_encoder.inverse_transform([predicted_class_index])[0]
        
        confidence = np.max(predictions) * 100
        
        return predicted_style, confidence
    except FileNotFoundError:
        return f"Error: Image file not found at {image_path}", None
    except Exception as e:
        return f"Error processing image {image_path}: {e}", None

# --- 5. Example Usage ---
if __name__ == "__main__":
    # First, load the model and label encoder
    classifier_model, classifier_label_encoder = load_classifier()

    if classifier_model and classifier_label_encoder:
        print("\n--- Demonstrating Prediction ---")
        # Create a dummy image for demonstration if a real image is not available
        demo_image_path = 'example_mayan_image.jpg'
        if not os.path.exists(demo_image_path):
            dummy_image = Image.fromarray(np.random.randint(0, 255, (224, 224, 3), dtype=np.uint8))
            dummy_image.save(demo_image_path)
            print(f"Created a dummy image: {demo_image_path}")
        
        # Replace 'demo_image_path' with the actual path to your image
        # e.g., 'path/to/your/new_image.jpg'
        
        predicted_style, confidence = predict_style(
            demo_image_path,
            classifier_model,
            classifier_label_encoder
        )

        if confidence is not None:
            print(f"Predicted Style for '{demo_image_path}': {predicted_style} (Confidence: {confidence:.2f}%) ")
        else:
            print(f"Prediction failed: {predicted_style}")

        # Clean up dummy image if created
        if os.path.exists(demo_image_path) and demo_image_path == 'example_mayan_image.jpg':
            os.remove(demo_image_path)
            print(f"Cleaned up dummy image: {demo_image_path}")
    else:
        print("Cannot proceed with prediction without a loaded model and label encoder.")
